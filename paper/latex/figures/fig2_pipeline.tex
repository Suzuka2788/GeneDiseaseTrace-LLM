% Figure 1 in the manuscript. The single-request pipeline. Blue = the one LLM call; grey = deterministic code.
% Counts: experiments/single_shot/README.md (100-paper run); word counts of the frozen V1 card/format.
% Drawn about 15.5 cm wide so it is scaled only slightly to the text width and small labels stay readable.
% Stage boxes hold top-aligned minipages of equal height so their headings line up.
\newcommand{\stagebox}[2]{\begin{minipage}[t][3.25cm][t]{#1}\raggedright #2\end{minipage}}
\newcommand{\codeline}{\ttfamily\fontsize{6.4pt}{7.6pt}\selectfont\color{muted}}
\begin{tikzpicture}[
  font=\sffamily\scriptsize, text=ink,
  every node/.append style={execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
  stage/.style={draw=rule, fill=panel, rounded corners=3pt, align=left, inner sep=4pt, anchor=north west},
  llm/.style={stage, draw=ours, fill=genefill, line width=0.8pt},
  io/.style={draw=rule, fill=white, rounded corners=2pt, text width=1.85cm, align=left, inner sep=3.5pt},
  note/.style={font=\sffamily\scriptsize},
]
% Steps (top-aligned, equal height)
\node[stage] (pack) at (2.75,0) {\stagebox{2.2cm}{%
  {\bfseries 1\enspace Packing}\par\smallskip
  Split into blocks and sentences with stable IDs; drop references.\par\smallskip
  {\codeline b0098.s12}\par\smallskip
  Card + format $\rightarrow$ cached system prompt.}};
\node[llm] (gen) at ($(pack.north east)+(0.45,0)$) {\stagebox{2.65cm}{%
  {\bfseries\color{ours} 2\enspace One LLM request}\par\smallskip
  Claude Opus 5.5, medium effort, no tools. Returns compact lines that cite sentence IDs only:\par\smallskip
  {\codeline E|G5|gene|MAP3K3|...\\EV|V7|b0098.s12|...\\P|G5|D3|sup|a|bg|...}}};
\node[stage] (exp) at ($(gen.north east)+(0.45,0)$) {\stagebox{2.2cm}{%
  {\bfseries 3\enspace Expansion}\par\smallskip
  Locate every surface form verbatim; resolve sentence IDs to quotes and offsets.\par\smallskip
  Build the full gene $\times$ disease matrix; unlisted pairs $\rightarrow$ {\ttfamily not\_found}.}};
\node[stage] (val) at ($(exp.north east)+(0.45,0)$) {\stagebox{2.2cm}{%
  {\bfseries 4\enspace Validation}\par\smallskip
  Surface forms occur in the text; sentence IDs exist; endpoints are primary entities; reason codes are legal.}};
% Inputs and output, centred on the step row
\path (pack.west) ++(-0.45,0) coordinate (inmid);
\node[io, anchor=south east] (paper) at ($(inmid)+(0,0.1)$) {\textbf{Paper}\\full text (PMC XML) or abstract};
\node[io, anchor=north east] (card) at ($(inmid)+(0,-0.1)$) {\textbf{Decision card}\\1,044 words + output format, 594 words};
\node[io, anchor=west] (out) at ($(val.east)+(0.6,0)$) {\textbf{Traceable output}\\entities, mentions\\evidence quotes\\4-status pair matrix\\bipartite graph};
% Arrows
\draw[flow] (paper.east) -- ++(0.2,0) |- ([yshift=3pt]pack.west);
\draw[flow] (card.east) -- ++(0.2,0) |- ([yshift=-3pt]pack.west);
\draw[flow] (pack) -- (gen);
\draw[flow] (gen) -- (exp);
\draw[flow] (exp) -- (val);
\draw[flow, draw=stsup] (val) -- node[above, note, text=stsup, font=\sffamily\scriptsize\bfseries] {pass} (out);
% Repair loop below the row
\draw[flow, draw=stneg, densely dashed, line width=1pt] (val.south) -- ++(0,-0.5) -| (gen.south);
\node[note, text=stneg, anchor=north, align=center] (fail) at ($(gen.south)!0.5!(val.south)+(0,-0.58)$)
  {\textbf{fail} $\rightarrow$ one repair request with the error list\\(31 of 100 papers; all were format errors)};
% Legend (left) and totals (right) on their own row below the repair note
\node[note, anchor=north west, text=muted] (leg) at ($(card.west |- fail.south)+(0,-0.15)$)
  {\textcolor{ours}{\rule{7pt}{7pt}}\,LLM call\quad \textcolor{rule!80!black}{\rule{7pt}{7pt}}\,deterministic code};
\node[note, anchor=north east, text=muted, align=right] at ($(out.east |- leg.north)$)
  {100-paper corpus: 69 pass on the first request, 100 after one repair;\\mean 78\,s and USD 0.44 per paper};
\end{tikzpicture}
